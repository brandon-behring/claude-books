% Four independently fallible evidence sources correlated for Ch24 ("Observability: Seeing What the Agent Did").
% Standalone TikZ; compiled by `book-scaffold build-figures` via pdflatex -> pdftocairo -> SVG.
% No single record is complete: transcript, telemetry, external effects, and billing/usage
% feed a correlation layer keyed by identifiers and time.
\documentclass[tikz,border=4mm]{standalone}
\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta, calc}

\begin{document}
\begin{tikzpicture}[
  font=\sffamily,
  correlate/.style={draw=blue!70!black, fill=blue!8, rounded corners=3pt, align=center,
    text width=5.0cm, minimum height=1.2cm, inner sep=4pt, font=\footnotesize},
  source/.style={draw=teal!70!black, fill=teal!10, rounded corners=3pt, align=center,
    text width=4.1cm, minimum height=1.35cm, inner sep=4pt, font=\footnotesize},
  flow/.style={-Stealth, very thick, draw=gray!55!black},
  note/.style={draw=gray!45!black, dash pattern=on 2pt off 2pt, rounded corners=2pt,
    align=center, inner sep=4pt, font=\sffamily\footnotesize, text=gray!35!black}
]

% --- independently owned evidence sources ---
\node[source] (transcript) at (-3.7,2.2)
  {\textbf{Session transcript}\\{\scriptsize messages, tool activity, metadata}};
\node[source] (telemetry) at (3.7,2.2)
  {\textbf{Runtime telemetry}\\{\scriptsize metrics, events, optional traces}};
\node[source] (effects) at (-3.7,-2.2)
  {\textbf{External effects}\\{\scriptsize git, CI, APIs, databases, audit logs}};
\node[source] (billing) at (3.7,-2.2)
  {\textbf{Billing \& usage}\\{\scriptsize provider or organization records}};

% --- the reconstruction is a join, not a hierarchy ---
\node[correlate] (join) at (0,0)
  {\textbf{Correlate the run}\\{\scriptsize session, request, tool, workflow IDs + time}};
\draw[flow] (transcript.south east) -- (join.north west);
\draw[flow] (telemetry.south west) -- (join.north east);
\draw[flow] (effects.north east) -- (join.south west);
\draw[flow] (billing.north west) -- (join.south east);

\node[note, text width=12.0cm] at (0,-3.45)
  {no single source is complete --- each is authoritative only for its own slice};

\end{tikzpicture}
\end{document}
